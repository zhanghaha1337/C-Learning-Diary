## 链表(单链表的插入)

顺序表靠连续数组存数据，占整块连续内存；链表不用连续内存，每个节点存数据 + 下个节点指针，零散节点靠指针串成逻辑有序的链式结构。

E = 数据类型这里是 int
一个节点 = 数据 + 指向下一个节点的指针
Node = 节点指针
```c
typedef int E;

struct listNode {
	E element;// 元素
	struct listNode* next;//指向下一个节点元素的指针;
};
```
初始化链表
创建一个空链表
把头节点的 next 设为 NULL
意思：后面没有节点，链表是空的
```c
void initList(Node node) {   //初始化函数
	node->next = NULL;
}
```
这个插入函数就是用来往链表指定位置加新节点的，首先先判断插入位置是不是太小，太小就直接插不了
接着循环挪动指针找到要插入位置的前一个节点，中途发现走到链表末尾就说明位置不对直接退出
开辟内存创建新节点并存放数据，先让新节点接上后面原本的节点，再让前面节点连上这个新节点，完成插入后就返回成功状态

node->next = head->next;
	head->next = node;
  这段举例原本链表顺序是 1、2、4、5、6，想把 3 插到 2 和 4 中间
  先让新节点 3 指向 4，再让节点 2 指向 3，这样就变成 1、2、3、4、5、6，数字不乱，也成功完成插入
```c
bool insertList(Node head,E element ,int index) {
	if (index < 1)return 0;//左边界
	while(--index){
		head = head->next;
		if (head == 0)return 0;//有边界如果head为0代表后面没有数据了
	}

	Node node = malloc(sizeof(struct listNode));//开辟空间,里面存折[数据|next指针]
	if (node == NULL)return 0;
	node->element = element;

	node->next = head->next;
	head->next = node;
	return 1;
}
```

创建一个头节点
初始化空链表
插入
打印
```c
int main() {
	struct listNode head;//创建一个头节点
	initList(&head);
	for (int i = 1; i <= 3; i++)
	{
		insertList(&head, i * 100, i);//需要从1开始,因为0是非法值
	}

	printfList(&head);
}
```

typedef int E;

struct listNode {
	E element;// 元素

typedef struct listNode* Node;

void initList(Node node) {   //初始化函数
	node->next = NULL;
}

bool insertList(Node head,E element ,int index) {
	if (index < 1)return 0;//左边界
	while(--index){
		head = head->next;
		if (head == 0)return 0;//有边界如果head为0代表后面没有数据了
	}

	Node node = malloc(sizeof(struct listNode));//开辟空间,里面存折[数据|next指针]
	if (node == NULL)return 0;
	node->element = element;

	node->next = head->next;
	head->next = node;
	return 1;
}
  
void printfList(Node head) {
	while (head->next){
		head = head->next;//跳过一个head开始遍历打印知道head.next==NULL返回0
		printf("%d ",head->element);
	}

}
