##链表增删改查

```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>


typedef int E;

struct listNode {
	E element;// 元素
	struct listNode* next;//指向下一个节点元素的指针;
};

typedef struct listNode* Node;

void initList(Node node) {   //初始化函数
	node->next = NULL;
}

bool insertList(Node head,E element ,int index) {
	if (index < 1)return 0;//左边界
	while(--index){
		head = head->next;
		if (head == 0)return 0;//有边界如果head为0代表后面没有数据了
	}

	Node node = malloc(sizeof(struct listNode));//开辟空间,里面存折[数据|next指针]
	if (node == NULL)return 0;
	node->element = element;

	node->next = head->next;
	head->next = node;
	return 1;
}
  
void printfList(Node head) {
	while (head->next){
		head = head->next;//跳过一个head开始遍历打印知道head.next==NULL返回0
		printf("%d ",head->element);
	}

}

bool deleList(Node head,int index) {
	if (index < 1)return 0;
	while (--index) {
		head = head->next;/*while (--index)这种写法可读性一般。可以写成for (int i = 1; i < index; i++)*/
		if (head == 0)return 0;
	}
	if (head->next == 0)return 0;
	
	Node tmp = head->next;
	head->next = head->next->next;
	free(tmp);

	return 1;
}

E* getList(Node head,int index) {
	if (index < 1)return 0;
	do{
	
		head = head->next;
		if (head == NULL)return 0;
	} while (--index);
	return &head->element;
}

int findList(Node head,E element) {
	head = head->next;
	int i = 1;
	while (head) {
		if (head->element == element)return i;
		head = head->next;
		i++;
	}
	return -1;
}

int sizeList(Node head) {
	int i = -1;
	while (head) {
		head = head->next;
		i++;
	}
	return i;
}


int main() {
	struct listNode head;//创建一个头节点
	initList(&head);
	for (int i = 1; i <= 3; i++)
	{
		insertList(&head, i * 100, i);//需要从1开始,因为0是非法值
	}

	printfList(&head);

	printf("------------------------\n");

	printf("%d", *getList(&head, 3));

	//deleList(&head, 2);
	printf("%d", sizeList(&head));
	//printfList(&head);
}
```
